const defaultState = { wallet: 840, savedAmount: 310, mealsLeft: 18, totalMeals: 24, activeKitchen: "Annapurna Homestyle Thali", tiffinDrop: "Campus Gate 2", tiffinSkipped: false, tomorrowRideCancelled: false, rides: [ { id: 1, pickup: "Hostel / PG Block", drop: "Campus Main Gate", time: "08:30", mode: "Shared Student Auto", fare: 45 } ] }; const cancelRideBtn = document.getElementById("cancelRideBtn"); if (state.tomorrowRideCancelled || state.rides.length === 0) { rideBadge.textContent = "Cancelled"; rideBadge.className = "status-skipped"; dashRideRoute.textContent = "No Morning Commute Locked"; dashRideMeta.textContent = "Book a 24-hr slot to lock 0% surge pricing"; cancelRideBtn.textContent = "Restore Morning Ride (₹45)"; } else { const nextRide = state.rides[0]; rideBadge.textContent = "Locked • 0% Surge"; rideBadge.className = "status-confirmed"; dashRideRoute.textContent = `${nextRide.pickup} → ${nextRide.drop}`; dashRideMeta.textContent = `Tomorrow, ${nextRide.time} • ${nextRide.mode} (₹${nextRide.fare})`; tiffinBadge.className = "status-skipped"; skipMealBtn.textContent = "Resume Tomorrow's Tiffin"; } else { tiffinBadge.textContent = "Scheduled"; tiffinBadge.className = "status-confirmed"; skipMealBtn.textContent = "Skip Meal (+₹90 Credit)"; } // Smart Sync Alert visibility const syncAlert = document.getElementById("syncAlert"); if (state.tomorrowRideCancelled && !state.tiffinSkipped && state.tiffinDrop.includes("Campus")) { syncAlert.classList.remove("hidden");     • `; } else { state.rides.forEach((ride) => { const li = document.createElement("li"); li.innerHTML = `

    Tomorrow at ${ride.time} • ${ride.mode} (₹${ride.fare})
    `; rideList.appendChild(li); }); } // Render Tiffin Progress in Tab 3 document.getElementById("punchLabel").textContent = `${state.mealsLeft} of ${state.totalMeals} meals remaining`; const pct = Math.round((state.mealsLeft / state.totalMeals) * 100); document.getElementById("punchFill").style.width = `${pct}%`; document.querySelectorAll('input[name="kitchen"]').forEach((radio) => { radio.checked = radio.value === state.activeKitchen; }); } // Cancel or Restore Tomorrow's Ride (Triggers the Commute-Tiffin Sync Banner) document.getElementById("cancelRideBtn").addEventListener("click", () => { if (!state.tomorrowRideCancelled && state.rides.length > 0) { state.tomorrowRideCancelled = true; state.wallet += 45; } else { state.tomorrowRideCancelled = false; state.wallet -= 45; if (state.rides.length === 0) { state.rides.push(defaultState.rides[0]); } } saveState(); }); // Switch Tiffin Drop Location (PG <-> Campus) document.getElementById("toggleDestinationBtn").addEventListener("click", () => { state.tiffinDrop = state.tiffinDrop === "Campus Gate 2" ?
